% Автоматическая сортировка номеров источников по возрастанию и сворачивание
% подряд идущих в диапазон: [3,1,2] -> [1--3] (sort и compress включены
% в пакете cite по умолчанию)
\usepackage[nospace]{cite}
\renewcommand{\citedash}{\textendash}% длинное тире вместо двойного дефиса

% Патч пакета cite: по умолчанию пакет сжимает в диапазон только
% последовательности из 3+ номеров ([5,6,7] -> [5-7]), но пары
% оставляет как [5,6]. Переопределяем \@compress@cite, чтобы пары
% также сворачивались: [5,6] -> [5-6].
\makeatletter
\def\@compress@cite#1#2#3{%
  \ifnum\@cite@incr=\z@%
    \advance\@tempcnta\@cclvi%
    \ifnum #1=\@tempcnta%
       \expandafter\def\expandafter\@h@ld\expandafter{\citedash\@cite@out{#3}}%
       \mathchardef\@cite@incr=\@cclvi%
    \else%
       \advance\@tempcnta-\@cclv%
       \ifnum #1=\@tempcnta%
         \expandafter\def\expandafter\@h@ld\expandafter{\citedash\@cite@out{#3}}%
         \mathchardef\@cite@incr=\@ne%
       \else%
         \@citea \@cite@out{#3}%
       \fi%
    \fi%
  \else%
    \advance\@tempcnta\@cite@incr%
    \ifnum #1=\@tempcnta%
       \def\@h@ld{\citedash \@cite@out{#3}}%
    \else%
       \@h@ld \@citea \@cite@out{#3}%
       \let\@h@ld\@empty%
       \mathchardef\@cite@incr=\z@%
    \fi%
  \fi%
  \@tempcnta#1\let\@citea\citepunct%
}
\makeatother

% Достройка границ рамки листинга на месте разрыва (используется в overlay
% окружения codelisting ниже). Это TikZ-команды, исполняются внутри overlay
% (tikzpicture) очередного куска tcolorbox, поэтому опираются на узел frame.
\newcommand{\listingbreaktop}{\draw[line width=0.5pt] (frame.north west) -- (frame.north east);}
\newcommand{\listingbreakbottom}{\draw[line width=0.5pt] (frame.south west) -- (frame.south east);}
\newcommand{\listingcontlabel}{\node[anchor=south west, inner sep=0pt,
      font=\normalfont\fontsize{14pt}{16pt}\selectfont]
      at ([yshift=2.5mm]frame.north west) {Продолжение листинга~\codelistingnumber};}

% codelisting: листинг с переносом через страницу. Использование: \begin{codelisting}{подпись}{label} \begin{minted}{lang}...\end{minted} \end{codelisting}
\newenvironment{codelisting}[2]{%
      \par\vspace{5pt}%
      \noindent\captionof{listing}{#1}\label{#2}%
      \edef\codelistingnumber{\thelisting}% запомнить номер для заголовка продолжения
      \nopagebreak[4]%
      \begin{tcolorbox}[%
                  breakable,%
                  enhanced,%
                  before skip=0pt,% рамка вплотную под подписью листинга (без отступа)
                  colback=white,%
                  colframe=black,%
                  arc=5pt,%
                  boxrule=0.5pt,%
                  left=4pt, right=5pt, top=5pt, bottom=5pt,%
                  % при переносе листинга на следующую страницу строка
                  % <<Продолжение листинга N>> печатается ВНЕ рамки, над
                  % продолжением (по аналогии с <<Продолжение таблицы>>):
                  % topsep at break резервирует место над рамкой продолжения,
                  % overlay рисует в нём подпись.
                  toprule at break=0.5pt, bottomrule at break=0.5pt,%
                  topsep at break=8mm,%
                  % tcolorbox 4.42 (enhanced) для verbatim-содержимого (minted)
                  % под XeLaTeX НЕ всегда рисует границу рамки на месте разрыва
                  % (то верхнюю у продолжения, то нижнюю у обрываемого куска):
                  % toprule/bottomrule at break заданы, но не срабатывают.
                  % Поэтому линии разрыва достраиваем вручную в overlay по
                  % геометрии frame.north/.south, чтобы рамка была замкнута на
                  % КАЖДОЙ странице переноса: верхняя линия — на кусках с разрывом
                  % сверху (middle, last), нижняя — на кусках с разрывом снизу
                  % (first, middle). Подпись <<Продолжение листинга>> — на кусках
                  % с разрывом сверху. Цельный кусок (unbroken) рисуется штатно.
                  overlay first={\listingbreakbottom},%
                  overlay middle={\listingcontlabel\listingbreaktop\listingbreakbottom},%
                  overlay last={\listingcontlabel\listingbreaktop},%
            ]%
            }{%
      \end{tcolorbox}%
      \vspace{5pt}%
}

\newcommand{\appendixchapter}[3]{%
  \chapter*{{\fontsize{14pt}{21pt}\selectfont\MakeUppercase{ПРИЛОЖЕНИЕ #1}}}%
  \phantomsection
  \addcontentsline{toc}{chapter}{ПРИЛОЖЕНИЕ #1}%
  \label{#3}%
  % Одинарный отступ между <<ПРИЛОЖЕНИЕ N>> и подзаголовком: гасим штатный
  % \GostAfterTitleSkip после \chapter* (локально, чтобы не трогать все главы),
  % оставляя зазор в одну строку. Подзаголовок остаётся жирным и центрированным.
  \vspace{-\GostAfterTitleSkip}%
  \begin{center}
    {\setstretch{1}\fontsize{14pt}{21pt}\selectfont\bfseries #2\par}
  \end{center}
}
