\begin{codelisting}{Псевдокод алгоритма NNH-TW с просмотром вперед}{listing:nnh_pseudocode}
  \begin{minted}{text}
Вход: граф G(V, E), startTimeUnix, constraints{startIdx, endIdx, precedences}
Выход: порядок обхода order[], тайминги[], агрегированная статистика

1.  prereqs <- buildPrereqs(precedences)
2.  endIdx <- constraints.endIdx (или -1)
3.  Если constraints.startIdx задан:
        cur <- constraints.startIdx
    Иначе:
        cur <- узел с наиболее ранним окном среди допустимых
4.  visited[cur] <- true;  order <- [cur]
5.  currentTimeSec <- startTimeUnix + duration[cur] * 60

6.  Пока |order| < |V|:
    // Если остался только закрепленный конечный узел — ставим его
    a. Если endIdx >= 0 И все прочие посещены:
         Добавить endIdx в маршрут; выйти из цикла

    // Проход 1: допустимый узел с минимальным completionTime + look-ahead
    b. bestComp <- +inf;  bestSafe <- false;  bestDeadline <- +inf;  next <- -1
    c. Для каждого i not in visited, i != endIdx:
         Если не выполнены предшественники i: пропустить
         arrivalSec <- currentTimeSec + travelSec[cur][i]
         Если НЕ feasible(i, arrivalSec): пропустить
         comp <- completionTime(i, arrivalSec)
         safe <- НЕ causesWindowMiss(i, comp, G, visited, endIdx)
         Если (safe > bestSafe) ИЛИ (safe = bestSafe И comp < bestComp)
              ИЛИ (safe = bestSafe И comp = bestComp И deadline[i] < bestDeadline):
             next <- i;  bestComp <- comp;  bestSafe <- safe

    // Проход 2: резервный — минимальный completionTime без учета окна
    d. Если next = -1:
         Для каждого i not in visited, i != endIdx:
             Если не выполнены предшественники i: пропустить
             comp <- completionTime(i, arrivalSec)
             Если comp < bestComp: next <- i

    e. waitSec <- max(0, windowStart[next] - arrivalSec)
    f. currentTimeSec <- arrivalSec + waitSec + duration[next] * 60
    g. visited[next] <- true;  order.append(next);  cur <- next

7.  Вернуть order, тайминги и статистику

feasible(i, arrivalSec):
    Если windowEnd[i] < 0: вернуть true
    Вернуть arrivalSec <= windowEnd[i] - duration[i] * 60

causesWindowMiss(cand, afterCandSec, G, visited, endIdx):
    Для каждого j not in visited, j != cand, j != endIdx:
        Если windowEnd[j] < 0: пропустить
        Если НЕ feasible(j, afterCandSec + travelSec[cand][j]):
            Вернуть true
    Вернуть false
  \end{minted}
\end{codelisting}

\begin{codelisting}{Реализация алгоритма ближайшего соседа с временными окнами и просмотром вперед, файл nearest\_neighbor.go}{app:lst_optimizer}
  \begin{minted}{go}
package optimizer

import (
    "context"
    "math"
)

// NearestNeighborTW — эвристика ближайшего соседа с учетом временных окон
// (NNH-TW). Все временные величины хранятся в секундах Unix. Сложность — O(n³).
type NearestNeighborTW struct{}

func NewNearestNeighborTW() *NearestNeighborTW { return &NearestNeighborTW{} }
func (a *NearestNeighborTW) Name() string      { return "nearest-neighbor-tw" }

// traversal инкапсулирует состояние обхода: текущая позиция, тайминги,
// агрегированные показатели; appendNode инкрементально его обновляет.
type traversal struct {
    g                                                          *Graph
    cur                                                        int
    currentTime                                                int64
    visited                                                    []bool
    order                                                      []int
    timings                                                    []StopTiming
    totalDistM, totalTravelSec, totalServiceSec, totalWaitSec  int
}

func (t *traversal) appendNode(next int) {
    e := t.g.Edges[t.cur][next]
    node := t.g.Nodes[next]
    arrivalSec := t.currentTime + int64(e.DurationSec)
    var waitSec int64
    if node.WindowStart >= 0 && arrivalSec < node.WindowStart {
        waitSec = node.WindowStart - arrivalSec
    }
    serviceStart := arrivalSec + waitSec
    serviceEnd := serviceStart + int64(node.DurationMin)*60

    t.totalDistM += e.DistanceM
    t.totalTravelSec += e.DurationSec
    t.totalServiceSec += node.DurationMin * 60
    t.totalWaitSec += int(waitSec)

    t.timings = append(t.timings, StopTiming{
        NodeIdx: next, ArrivalSec: arrivalSec, WaitSec: int(waitSec),
        ServiceStartSec: serviceStart, ServiceEndSec: serviceEnd,
        TravelFromPrevSec: e.DurationSec,
    })
    t.visited[next] = true
    t.order = append(t.order, next)
    t.cur = next
    t.currentTime = serviceEnd
}

func (a *NearestNeighborTW) Optimize(
    _ context.Context, g *Graph, startTimeUnix int64, c Constraints,
) (Result, error) {
    n := len(g.Nodes)
    if n == 0 {
        return Result{}, nil
    }
    prereqs := buildPrereqs(n, c.PrecedencePairs)
    endIdx := -1
    if c.EndNodeIdx != nil {
        endIdx = *c.EndNodeIdx
    }

    cur := 0
    if c.StartNodeIdx != nil {
        cur = *c.StartNodeIdx
    } else {
        cur = startNode(g, prereqs, endIdx)
    }
    startServiceEnd := startTimeUnix + int64(g.Nodes[cur].DurationMin)*60
    t := &traversal{
        g: g, cur: cur, currentTime: startServiceEnd,
        visited: make([]bool, n), order: []int{cur},
        timings: []StopTiming{{
            NodeIdx: cur, ArrivalSec: startTimeUnix,
            ServiceStartSec: startTimeUnix, ServiceEndSec: startServiceEnd,
        }},
        totalServiceSec: g.Nodes[cur].DurationMin * 60,
    }
    t.visited[cur] = true

    for len(t.order) < n {
        // Если остался только закрепленный конечный узел — ставим его и выходим.
        if endIdx >= 0 && !t.visited[endIdx] && onlyEndUnvisited(t.visited, endIdx) {
            t.appendNode(endIdx)
            break
        }
        next := pickNextFeasible(t, prereqs, endIdx)
        if next == -1 {
            next = pickNextFallback(t, prereqs, endIdx)
        }
        if next == -1 {
            // Циклический граф предшествования — конечный узел добавим и завершим.
            if endIdx >= 0 && !t.visited[endIdx] {
                t.appendNode(endIdx)
            }
            break
        }
        t.appendNode(next)
    }
    return Result{
        Order: t.order, Timings: t.timings,
        TotalDistanceM: t.totalDistM, TotalTravelSec: t.totalTravelSec,
        TotalServiceSec: t.totalServiceSec, TotalWaitSec: t.totalWaitSec,
    }, nil
}

// pickNextFeasible — проход 1: допустимый узел с минимальным временем
// завершения + look-ahead. Безопасные кандидаты предпочитаются.
func pickNextFeasible(t *traversal, prereqs [][]int, endIdx int) int {
    g := t.g
    next := -1
    var bestComp int64 = math.MaxInt64
    var bestDeadline int64 = math.MaxInt64
    bestSafe := false
    for i := range g.Nodes {
        if t.visited[i] || i == endIdx || !prereqsMet(i, t.visited, prereqs) {
            continue
        }
        arrival := t.currentTime + int64(g.Edges[t.cur][i].DurationSec)
        if !feasible(g.Nodes[i], arrival) {
            continue
        }
        comp := nodeCompletionTime(g.Nodes[i], arrival)
        deadline := g.Nodes[i].WindowEnd
        if deadline < 0 {
            deadline = math.MaxInt64
        }
        safe := !causesWindowMiss(i, comp, g, t.visited, endIdx)
        better := false
        switch {
        case safe && !bestSafe:
            better = true
        case safe == bestSafe:
            better = comp < bestComp ||
                (comp == bestComp && deadline < bestDeadline)
        }
        if better {
            bestComp = comp; bestDeadline = deadline
            bestSafe = safe; next = i
        }
    }
    return next
}

// pickNextFallback — проход 2: минимальный completionTime, окно игнорируется,
// предшествование сохраняется.
func pickNextFallback(t *traversal, prereqs [][]int, endIdx int) int {
    g := t.g
    next := -1
    var bestComp int64 = math.MaxInt64
    for i := range g.Nodes {
        if t.visited[i] || i == endIdx || !prereqsMet(i, t.visited, prereqs) {
            continue
        }
        arrival := t.currentTime + int64(g.Edges[t.cur][i].DurationSec)
        comp := nodeCompletionTime(g.Nodes[i], arrival)
        if comp < bestComp {
            bestComp = comp; next = i
        }
    }
    return next
}

func feasible(node Node, arrivalSec int64) bool {
    if node.WindowEnd < 0 {
        return true
    }
    return arrivalSec <= node.WindowEnd-int64(node.DurationMin)*60
}

// causesWindowMiss — look-ahead на один шаг: true, если после cand
// хотя бы один узел с окном становится недостижимым.
func causesWindowMiss(cand int, afterSec int64,
    g *Graph, visited []bool, endIdx int) bool {
    for j := 0; j < len(g.Nodes); j++ {
        if visited[j] || j == cand || j == endIdx {
            continue
        }
        if g.Nodes[j].WindowEnd < 0 {
            continue
        }
        arr := afterSec + int64(g.Edges[cand][j].DurationSec)
        if !feasible(g.Nodes[j], arr) {
            return true
        }
    }
    return false
}

// requestWithAuth выполняет HTTP-запрос с токеном доступа.
// При получении 401 однократно обновляет токен и повторяет запрос.
async function requestWithAuth<T>(
  path: string, init: RequestInit,
  options: AuthorizedRequestOptions,
): Promise<T> {
  try {
    return await request<T>(path, init, options.session.accessToken);
  } catch (error) {
    if (!(error instanceof ApiError) || error.status !== 401) throw error;
  }
  try {
    const nextSession = await refreshSession(options);
    return request<T>(path, init, nextSession.accessToken);
  } catch (error) {
    await options.onUnauthorized();
    throw error;
  }
}

// refreshSession — дедупликация через единственный промис.
async function refreshSession(options: AuthorizedRequestOptions) {
  if (!refreshSessionPromise) {
    refreshSessionPromise = request<TokenPairResponse>('/auth/refresh', {
      method: 'POST',
      body: JSON.stringify({
        refreshToken: options.session.refreshToken,
      }),
    })
      .then((pair) => {
        const nextSession = buildSession(pair,
          getEmailFromToken(pair.accessToken) ?? options.session.email);
        options.onSessionChange(nextSession);
        return nextSession;
      })
      .finally(() => { refreshSessionPromise = null; });
  }
  return refreshSessionPromise;
}

\begin{codelisting}{Построение матрицы расстояний через Яндекс Маршруты JS API, файл routeOptimizer.ts}{app:lst_route_optimizer}

import { Task } from '../types/task';
import { loadYandexMaps } from './yandexMaps';
import { roundUpSecToMinute } from './time';

export interface DistanceCell {
  distanceM: number;
  durationSec: number;
}

/**
 * Строит матрицу расстояний n×n для списка задач через Яндекс Маршруты.
 * Используется тот же движок, что и для визуализации маршрута на карте,
 * поэтому данные оптимизации согласованы с отображением.
 *
 * matrix[i][j] = стоимость проезда от tasks[i] до tasks[j].
 * Диагональные элементы (i === j) равны нулю.
 */
export async function buildYandexDistanceMatrix(
  tasks: Task[],
  routingMode: 'auto' | 'pedestrian' | 'masstransit' = 'auto',
): Promise<DistanceCell[][]> {
  const n = tasks.length;
  if (n === 0) return [];

  await loadYandexMaps();

  // masstransit через multiRouter не возвращает надежные данные по времени —
  // используем 'auto' как приближение.
  const mode = routingMode === 'masstransit' ? 'auto' : routingMode;

  const fallbackSec = 99_999;
  const matrix: DistanceCell[][] = Array.from({ length: n }, (_, i) =>
    Array.from({ length: n }, (__, j) =>
      i === j
        ? { distanceM: 0, durationSec: 0 }
        : { distanceM: 0, durationSec: fallbackSec },
    ),
  );

  // Запрашиваем все пары параллельно: n*(n-1) запросов.
  const pairs: [number, number][] = [];
  for (let i = 0; i < n; i++)
    for (let j = 0; j < n; j++)
      if (i !== j) pairs.push([i, j]);

  await Promise.all(
    pairs.map(async ([i, j]) => {
      const from = tasks[i];
      const to = tasks[j];
      if (from.latitude == null || to.latitude == null) return;

      try {
        const { distanceM, durationSec } = await new Promise<{
          distanceM: number;
          durationSec: number;
        }>((resolve, reject) => {
          const mr = new (window.ymaps as any).multiRouter.MultiRoute(
            {
              referencePoints: [
                [from.latitude!, from.longitude!],
                [to.latitude!, to.longitude!],
              ],
              params: { routingMode: mode },
            },
            {},
          );

          mr.model.events.once('requestsuccess', () => {
            try {
              const activeRoute: any = mr.getActiveRoute();
              if (activeRoute) {
                const distObj: any = activeRoute.properties.get('distance');
                const durObj: any = activeRoute.properties.get('duration');
                resolve({
                  distanceM: Math.round(
                    typeof distObj?.value === 'number' ? distObj.value : 0,
                  ),
                  durationSec: roundUpSecToMinute(
                    typeof durObj?.value === 'number'
                      ? durObj.value
                      : fallbackSec,
                  ),
                });
                return;
              }

              reject(new Error('no route data'));
            } catch (e) {
              reject(e);
            }
          });

          mr.model.events.once('requestfail', () => {
            reject(new Error('routing failed'));
          });
        });

        matrix[i][j] = { distanceM, durationSec };
      } catch {
        // Оставляем fallback: алгоритм избегает недостижимых пар.
      }
    }),
  );

  return matrix;
}
